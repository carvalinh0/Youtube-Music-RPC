function renderWidget({ title, artist, current_time_in_seconds, total_time_in_seconds, is_playing, formatTime }) {
  const root = document.getElementById('widget-root');
  if (!title) { root.innerHTML = ''; return; }

  let styleEl = document.getElementById('brutalist-widget-styles');
  if (!styleEl) {
    styleEl = document.createElement('style');
    styleEl.id = 'brutalist-widget-styles';
    styleEl.textContent = `
      @import url('https://fonts.googleapis.com/css2?family=Lexend+Mega:wght@800&family=Space+Mono:wght@700&display=swap');
      
      :root {
        --brutal-font-title: 'Lexend Mega', sans-serif;
        --brutal-font-mono: 'Space Mono', monospace;
        --color-yellow: #ffde4d;
        --color-pink: #ff007f;
        --color-mint: #00ffcc;
        --color-blue: #3b82f6;
        --color-black: #000000;
        --color-white: #ffffff;
      }

      .brutal-card {
        font-family: var(--brutal-font-mono);
        background: var(--color-yellow);
        color: var(--color-black);
        border: 4px solid var(--color-black);
        border-radius: 4px;
        padding: 16px;
        width: 380px;
        box-shadow: 6px 6px 0px var(--color-black);
        display: flex;
        flex-direction: column;
        gap: 14px;
        box-sizing: border-box;
        position: relative;
      }

      .brutal-badge {
        position: absolute;
        top: -12px;
        left: 14px;
        background: var(--color-pink);
        color: var(--color-white);
        border: 3px solid var(--color-black);
        padding: 2px 8px;
        font-family: var(--brutal-font-title);
        font-size: 8px;
        text-transform: uppercase;
        letter-spacing: 1px;
        box-shadow: 2px 2px 0px var(--color-black);
        z-index: 10;
      }

      .brutal-header {
        display: flex;
        align-items: center;
        justify-content: space-between;
        gap: 12px;
        margin-top: 4px;
      }

      .song-title-wrap {
        flex: 1;
        overflow: hidden;
        white-space: nowrap;
        background: var(--color-white);
        border: 3px solid var(--color-black);
        padding: 8px;
        box-shadow: inset 3px 3px 0px rgba(0,0,0,0.1);
      }

      .song-title {
        font-family: var(--brutal-font-title);
        font-size: 13px;
        color: var(--color-black);
        display: inline-block;
        text-transform: uppercase;
      }

      .marquee-active {
        animation: brutalMarquee 10s linear infinite;
        padding-right: 40px;
      }

      @keyframes brutalMarquee {
        0% { transform: translate3d(0, 0, 0); }
        100% { transform: translate3d(-50%, 0, 0); }
      }

      .brutal-heart {
        font-size: 24px;
        color: var(--color-pink);
        display: flex;
        align-items: center;
        justify-content: center;
        animation: heartPulse 0.8s ease infinite alternate;
        animation-play-state: paused;
        user-select: none;
      }

      .is-playing .brutal-heart {
        animation-play-state: running;
      }

      @keyframes heartPulse {
        0% { transform: scale(1); filter: drop-shadow(0px 0px 0px #000); }
        100% { transform: scale(1.25); filter: drop-shadow(2px 2px 0px #000); }
      }

      .song-artist {
        font-size: 12px;
        font-weight: 700;
        color: var(--color-black);
        text-transform: uppercase;
        letter-spacing: -0.5px;
        white-space: nowrap;
        overflow: hidden;
        text-overflow: ellipsis;
        background: var(--color-mint);
        border: 3px solid var(--color-black);
        padding: 4px 8px;
        align-self: flex-start;
      }

      .brutal-progress-section {
        display: flex;
        flex-direction: column;
        gap: 8px;
      }

      .brutal-progress-track {
        height: 18px;
        background: var(--color-white);
        border: 3px solid var(--color-black);
        position: relative;
        overflow: hidden;
        box-shadow: inset 3px 3px 0px rgba(0,0,0,0.15);
      }

      /* Listras diagonais de perigo com animação deslizante */
      .brutal-progress-fill {
        height: 100%;
        background: repeating-linear-gradient(
          -45deg,
          var(--color-pink),
          var(--color-pink) 10px,
          var(--color-blue) 10px,
          var(--color-blue) 20px
        );
        background-size: 40px 40px;
        border-right: 3px solid var(--color-black);
        transition: width 0.5s linear;
        animation: moveStripes 1s linear infinite;
        animation-play-state: paused;
      }

      .is-playing .brutal-progress-fill {
        animation-play-state: running;
      }

      @keyframes moveStripes {
        0% { background-position: 0 0; }
        100% { background-position: 40px 0; }
      }

      .brutal-time-row {
        display: flex;
        justify-content: space-between;
        font-size: 11px;
        font-weight: 700;
        text-transform: uppercase;
      }

      .time-badge {
        background: var(--color-white);
        border: 2px solid var(--color-black);
        padding: 2px 6px;
        border-radius: 2px;
      }
    `;
    document.head.appendChild(styleEl);
  }

  const pct = total_time_in_seconds > 0
    ? (current_time_in_seconds / total_time_in_seconds * 100).toFixed(2)
    : 0;

  const wrapper = root.querySelector('.brutal-card');
  const prevTitle = wrapper ? wrapper._songTitle : null;
  const prevArtist = wrapper ? wrapper._songArtist : null;

  if (!wrapper || prevTitle !== title || prevArtist !== artist) {
    const isTitleLong = title.length > 22;
    const titleDisplay = isTitleLong 
      ? `${title} &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp; ${title}` 
      : title;

    root.innerHTML = `
      <div class="brutal-card ${is_playing ? 'is-playing' : ''}">
        <div class="brutal-badge">NOW PLAYING</div>
        
        <div class="brutal-header">
          <div class="song-title-wrap">
            <span class="song-title ${isTitleLong ? 'marquee-active' : ''}">${titleDisplay}</span>
          </div>
          <div class="brutal-heart">♥</div>
        </div>

        <div class="song-artist">${artist}</div>

        <div class="brutal-progress-section">
          <div class="brutal-progress-track">
            <div class="brutal-progress-fill" style="width: ${pct}%"></div>
          </div>
          <div class="brutal-time-row">
            <span class="time-badge curr-time">${formatTime(current_time_in_seconds)}</span>
            <span class="time-badge">${formatTime(total_time_in_seconds)}</span>
          </div>
        </div>
      </div>
    `;

    const newWrapper = root.querySelector('.brutal-card');
    if (newWrapper) {
      newWrapper._songTitle = title;
      newWrapper._songArtist = artist;
    }
  } else {
    if (is_playing) {
      wrapper.classList.add('is-playing');
    } else {
      wrapper.classList.remove('is-playing');
    }

    const progressFill = wrapper.querySelector('.brutal-progress-fill');
    if (progressFill) {
      progressFill.style.width = `${pct}%`;
    }

    const currTimeEl = wrapper.querySelector('.curr-time');
    if (currTimeEl) {
      currTimeEl.textContent = formatTime(current_time_in_seconds);
    }
  }
}
